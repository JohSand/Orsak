GET /orders/{orderId:int}
  operationId: getOrder
  tag: Orders
  summary: Gets an order, with its lines.
  description: The lines are in the order they were added. See OrderLine.
  parameter orderId, in path, integer (int32): The id of the order.
  success: Order
  success description: The order.
  error: OrderError
    case OrderNotFound of orderId: Int32: No order has the id.
    case LineNotFound of orderId: Int32, line: Int32: The order has no line with the number.
    case Forbidden: The caller may not see the order.
  response 404: No order has the id.
  environment: Unit
